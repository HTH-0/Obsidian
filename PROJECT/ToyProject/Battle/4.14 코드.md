// Battle.java
public class Battle {

    public void battle(Player player, Enemy enemy, BattleChoice choice) {
        switch(choice) {
        case ATTACK:
            attack(player, enemy);
            break;
        case DEFEND:
            defend(player);
            break;
        case MAGIC:
            magic();
            break;
        }
    }

    private void attack(Player player, Enemy enemy) {
        int damage = player.getPower() - enemy.getDefense();
        enemy.setHp(enemy.getHp() - damage);
        System.out.printf("%s에게 %d의 피해를 입히였습니다!\n 남은 체력 %d\n", enemy.getName(), damage, enemy.getHp());
    }

    private void defend(Player player) {
        player.setNewDefense(player.getDefense() * 2);
        System.out.println("플레이어가 방어 태세를 치합니다. 방어력 2배!");
    }

    private void magic() {
        System.out.println("아직 마법 기능은 구현되지 않았습니다.");
    }
}

// Enemy.java
public class Enemy {

    private String name;
    private int hp;
    private int defense;
    private int power;

    public Enemy(String name, int hp, int defense, int power) {
        super();
        this.name = name;
        this.hp = hp;
        this.defense = defense;
        this.power = power;
    }

    public String getName() { return name; }
    public void setName(String name) { this.name = name; }
    public int getHp() { return hp; }
    public void setHp(int hp) { this.hp = hp; }
    public int getDefense() { return defense; }
    public void setDefense(int defense) { this.defense = defense; }
    public int getPower() { return power; }
    public void setPower(int power) { this.power = power; }

    public Enemy copy() {
        return new Enemy(name, hp, defense, power);
    }

    @Override
    public String toString() {
        return String.format("%s (HP: %d, ATK: %d, DEF: %d)", name, hp, power, defense);
    }

    public static final Enemy enemyA = new Enemy("enemyA", 100, 10, 10);
    public static final Enemy enemyB = new Enemy("enemyB", 200, 20, 20);
    public static final Enemy enemyC = new Enemy("enemyC", 500, 30, 30);
    public static final Enemy enemyD = new Enemy("enemyD", 1000, 40, 40);
}

// Player.java
public class Player {
    private int hp;
    private int mana;
    private final int defense;
    private int power;
    private int newDefense;

    public Player(int hp, int mana, int defense, int power) {
        super();
        this.hp = hp;
        this.mana = mana;
        this.defense = defense;
        this.newDefense = defense;
        this.power = power;
    }

    public int getHp() { return hp; }
    public void setHp(int hp) { this.hp = hp; }
    public int getMana() { return mana; }
    public void setMana(int mana) { this.mana = mana; }
    public int getDefense() { return defense; }
    public int getNewDefense() { return newDefense; }
    public void setNewDefense(int newDefense) { this.newDefense = newDefense; }
    public int getPower() { return power; }
    public void setPower(int power) { this.power = power; }
}

// Stage.java
import java.util.ArrayList;
import java.util.List;

public class Stage {
    private String stageNumber;
    private List<Enemy> enemyList;

    public Stage(String stageNumber, List<Enemy> enemyList) {
        this.stageNumber = stageNumber;
        this.enemyList = new ArrayList<>(enemyList);
    }

    public String getStageNumber() { return stageNumber; }
    public List<Enemy> getEnemyList() { return enemyList; }

    @Override
    public String toString() {
        return "[%s] - 적 수: %d명".formatted(stageNumber, enemyList.size());
    }
}
